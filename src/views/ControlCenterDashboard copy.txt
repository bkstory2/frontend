<template>
  <div class="dashboard">
    <div ref="chartEl" class="chart"></div>
  </div>
</template>

<script>
import * as echarts from "echarts"
import axios from "axios"

export default {
  name: "ControlCenterDashboard",
  data() {
    return {
      chart: null
    }
  },
  async mounted() {
    this.chart = echarts.init(this.$refs.chartEl)
    await this.loadData()
    window.addEventListener("resize", this.resize)
  },
  beforeUnmount() {
    window.removeEventListener("resize", this.resize)
    if (this.chart) this.chart.dispose()
  },
  methods: {
    async loadData() {
      let raw = [];
      try {
        const res = await axios.get("/api/stats");
        raw = Array.isArray(res.data) ? res.data : [];
      } catch (e) {
        // fallback 샘플
        raw = [
          { API_CALL_CD: "002", ACCS_URL_PATH: "/order", HTTP_STACD_VAL: "200", callCnt: 203, avgDurationSec: 2.6 },
          { API_CALL_CD: "003", ACCS_URL_PATH: "/logistics", HTTP_STACD_VAL: "200", callCnt: 596, avgDurationSec: 0.3 },
          { API_CALL_CD: "999", ACCS_URL_PATH: "/pkg/baseInfo", HTTP_STACD_VAL: "400", callCnt: 85, avgDurationSec: 4.2 }
        ];
      }
      this.render(raw);
    },
    render(raw) {
      // 캔버스 좌표계로 3개 지점과 서버 배치 (넓게)
      const width = 1200, height = 700;
      const apiServer = { name: "API 서버", coord: [width/2, height/2] };
      const apiMap = {
        "002": { name: "서울", coord: [width/4, height/4] },
        "003": { name: "도쿄", coord: [width*3/4, height/4] },
        "999": { name: "베이징", coord: [width/2, height*3/4] }
      };
      // 라인 데이터 (많이 보이게 랜덤 생성)
      let lines = [];
      for (let k of Object.keys(apiMap)) {
        for (let i = 0; i < 80; i++) {
          const from = apiMap[k].coord;
          const to = apiServer.coord;
          const ctrl = [
            from[0] + (Math.random() - 0.5) * 120,
            from[1] + (Math.random() - 0.5) * 120
          ];
          lines.push({
            coords: [from, ctrl, to],
            lineStyle: {
              color: "#00eaff",
              width: 1 + Math.random() * 2,
              opacity: 0.7 + Math.random() * 0.3
            }
          });
        }
      }
      // 실제 데이터 라인 강조
      lines = lines.concat(raw.map(r => ({
        coords: [apiMap[r.API_CALL_CD]?.coord, apiServer.coord],
        lineStyle: {
          color: r.HTTP_STACD_VAL === "200" ? "#00eaff" : "#ff2e63",
          width: 4,
          opacity: 1
        }
      })).filter(l => l.coords[0] && l.coords[1]));

      // 포인트 데이터
      const points = Object.values(apiMap).concat([apiServer]);

      this.chart.setOption({
        backgroundColor: "#181c24",
        xAxis: { show: false, min: 0, max: width },
        yAxis: { show: false, min: 0, max: height, inverse: true },
        grid: { left: 0, right: 0, top: 0, bottom: 0 },
        series: [
          {
            type: "lines",
            coordinateSystem: "cartesian2d",
            polyline: true,
            effect: {
              show: true,
              period: 2,
              trailLength: 0.7,
              color: "#00eaff",
              symbolSize: 4
            },
            lineStyle: {
              width: 1,
              opacity: 0.6,
              curveness: 0.3
            },
            data: lines
          },
          {
            type: "effectScatter",
            coordinateSystem: "cartesian2d",
            symbolSize: 28,
            itemStyle: { color: "#ffeb3b", borderColor: "#fff", borderWidth: 2 },
            label: {
              show: true,
              formatter: p => points[p.dataIndex].name,
              color: "#fff",
              fontSize: 18,
              backgroundColor: "rgba(0,0,0,0.5)",
              padding: [2, 8],
              position: "top"
            },
            data: points.map(a => ({ value: a.coord, name: a.name }))
          }
        ]
      });
    },

    resize() {
      if (this.chart) this.chart.resize()
    }
  }
}
</script>

.dashboard {
  width: 100vw;
  height: 100vh;
  background: #181c24;
  position: relative;
}
.dashboard::before {
  content: "";

  left: 0; top: 0; right: 0; bottom: 0;

  opacity: 0.18;
  z-index: 0;
  pointer-events: none;
}
.chart {
  width: 100%;
  height: 100%;
  position: relative;
  z-index: 1;
}
